cmake_minimum_required(VERSION 3.20)
project(AstrelisWindow LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows only")
endif()
add_library(window_outline SHARED outline.cpp win_delay_load_hook.cc)
target_compile_features(window_outline PRIVATE cxx_std_17)
target_include_directories(window_outline PRIVATE "${NODE_INCLUDE_DIR}")
target_compile_definitions(window_outline PRIVATE BUILDING_NODE_EXTENSION HOST_BINARY="node.exe")
target_link_libraries(window_outline PRIVATE "${NODE_LIBRARY}" user32 gdi32 dwmapi delayimp)
target_link_options(window_outline PRIVATE /DELAYLOAD:node.exe)
set_target_properties(window_outline PROPERTIES PREFIX "" SUFFIX ".node" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
